## §1. Stage enumeration, from the code, across the whole pipeline

**Probe** (AST walk over BOTH `typsphinx/template_engine.py` and `typsphinx/writer.py`, tracking
the enclosing `FunctionDef` for every node via an explicit stack, collecting every `Assign` whose
target is a `Subscript` on `Name(id in {"params", "all_params"})` and every `.update(...)` `Call`
whose receiver is such a `Name`):

```python
import ast
from pathlib import Path

TEMPLATE_ENGINE_PATH = Path('typsphinx/template_engine.py')
WRITER_PATH = Path('typsphinx/writer.py')
PARAMS_CARRIERS = ('params', 'all_params')


class _StageSiteVisitor(ast.NodeVisitor):
    def __init__(self):
        self.func_stack = []
        self.findings = []  # (func, kind, slice, lineno)

    def visit_FunctionDef(self, node):
        self.func_stack.append(node.name)
        self.generic_visit(node)
        self.func_stack.pop()

    def visit_Assign(self, node):
        if self.func_stack:
            func_name = self.func_stack[-1]
            for target in node.targets:
                if not (
                    isinstance(target, ast.Subscript)
                    and isinstance(target.value, ast.Name)
                    and target.value.id in PARAMS_CARRIERS
                ):
                    continue
                slice_node = target.slice
                if isinstance(slice_node, ast.Constant) and isinstance(slice_node.value, str):
                    slice_repr = slice_node.value
                elif isinstance(slice_node, ast.Name):
                    slice_repr = slice_node.id
                else:
                    slice_repr = ast.unparse(slice_node)
                self.findings.append((func_name, 'assign', slice_repr, node.lineno))
        self.generic_visit(node)

    def visit_Call(self, node):
        if self.func_stack:
            func_name = self.func_stack[-1]
            if (
                isinstance(node.func, ast.Attribute)
                and node.func.attr == 'update'
                and isinstance(node.func.value, ast.Name)
                and node.func.value.id in PARAMS_CARRIERS
                and node.args
            ):
                arg_repr = ast.unparse(node.args[0])
                self.findings.append((func_name, 'update', arg_repr, node.lineno))
        self.generic_visit(node)


for path in (TEMPLATE_ENGINE_PATH, WRITER_PATH):
    tree = ast.parse(path.read_text(encoding='utf-8'))
    visitor = _StageSiteVisitor()
    visitor.visit(tree)
    print(f"--- {path} ---")
    for func_name, kind, slice_repr, lineno in visitor.findings:
        print(f"  line {lineno:4d}  {path}::{func_name}::{kind}::{slice_repr}")
```

## §2. The generated falsification row space, widened along the STAGE axis

**Probe** (cross product of mapping SOURCE key x mapping TARGET key x source-PRESENT/absent x

```python
import itertools
from typsphinx.template_engine import TemplateEngine

SEED_NAME = 'Jane Doe'
TYPST_AUTHORS = {SEED_NAME: {'organization': 'MIT'}}
SEED_AFTER_STAGE1 = [{'name': SEED_NAME, 'organization': 'MIT'}]
TF_AUTHOR = 'FROM TEMPLATE FUNCTION'
BASE_METADATA = {'project': 'P', 'author': 'Explicit Entry Author', 'release': '1.0'}
SOURCE_KEYS = ('author', 'project', 'release')
TARGET_KEYS = ('authors', 'doc_authors', 'writer')

rows = []
for source, target, present, tf in itertools.product(SOURCE_KEYS, TARGET_KEYS, (True, False), (True, False)):
    mapping = {source: target}
    metadata = dict(BASE_METADATA) if present else {k: v for k, v in BASE_METADATA.items() if k != source}
    typst_template_function = {'name': 'project', 'params': {'authors': (TF_AUTHOR,)}} if tf else None
    engine = TemplateEngine(
        parameter_mapping=mapping,
        typst_authors=TYPST_AUTHORS,
        typst_template_function=typst_template_function,
    )
    params = engine.map_parameters(metadata)
    survived_stage1 = params.get('authors') == SEED_AFTER_STAGE1
    rendered = engine.render(params, body='BODY_MARKER')
    seed_in_render = SEED_NAME in rendered
    row_id = f"{source}-to-{target}-{'present' if present else 'absent'}-{'tf' if tf else 'notf'}"
    rows.append(dict(
        row_id=row_id, source=source, target=target, present=present, tf=tf,
        mapping=mapping, authors_after_stage1=params.get('authors'),
        survived_stage1=survived_stage1, seed_in_rendered=seed_in_render,
    ))

for r in rows:
    print(r['row_id'], '| mapping=', r['mapping'], '| present=', r['present'],
          '| tf=', r['tf'], '| survived_stage1=', r['survived_stage1'],
          '| authors_after_stage1=', r['authors_after_stage1'],
          '| seed_name_in_rendered=', r['seed_in_rendered'])
print()
print("Total rows:", len(rows))
```
